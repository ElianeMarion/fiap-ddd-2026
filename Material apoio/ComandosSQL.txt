CREATE TABLE TBL_CONTATO(
    ID_CONTATO NUMERIC(4) PRIMARY KEY,

    CELULAR_CONTATO VARCHAR2(11) NOT NULL,
    EMAIL_CONTATO VARCHAR2(60) NOT NULL,
    INSTAGRAM VARCHAR2(50),
